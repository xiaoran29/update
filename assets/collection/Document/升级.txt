并发处理能力

```python
# 现有：单线程顺序处理
for url in urls:
    process_url(url)  # 一个一个来

# 升级：多线程/异步处理
async def process_all_urls(urls):
    tasks = [process_url(url) for url in urls]
    await asyncio.gather(*tasks)  # 同时处理
```

```python

def process_url(url):
    if cache.exists(url) and not expired(url):
        use_cache(url)  # 用缓存
    else:
        data = download_and_process(url)
        save_cache(url, data)  # 保存缓存
```

```python
# 第一步：添加简单的并发处理
import concurrent.futures

def process_urls_concurrently(urls, max_workers=10):
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {executor.submit(process_url, url): url for url in urls}
        for future in concurrent.futures.as_completed(futures):
            try:
                future.result()
            except Exception as e:
                print(f"处理失败: {e}")
```
