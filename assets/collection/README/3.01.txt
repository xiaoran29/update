```python
# 插件基类（抽象接口）
from abc import ABC, abstractmethod
from typing import Dict, List, Optional, Any

class Plugin(ABC):
    """所有插件的基类"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.name = self.__class__.__name__
        self.version = "1.0.0"
    
    @abstractmethod
    def initialize(self) -> bool:
        """插件初始化"""
        pass
    
    @abstractmethod
    def process(self, data: Any, context: Dict[str, Any]) -> Any:
        """处理数据"""
        pass
    
    @abstractmethod
    def cleanup(self) -> None:
        """清理资源"""
        pass
    
    def on_error(self, error: Exception) -> bool:
        """错误处理（可重写）"""
        print(f"插件 {self.name} 发生错误: {error}")
        return False  # 默认停止处理

# 插件管理器（工厂模式）
class PluginManager:
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.plugins: Dict[str, Plugin] = {}
        self.load_order: List[str] = []
    
    def load_plugins(self) -> None:
        """动态加载所有启用的插件"""
        plugin_configs = self.config.get('plugins', {})
        
        for plugin_name, plugin_config in plugin_configs.items():
            if plugin_config.get('enabled', False):
                plugin_class = self._get_plugin_class(plugin_name)
                if plugin_class:
                    plugin = plugin_class(plugin_config)
                    if plugin.initialize():
                        self.plugins[plugin_name] = plugin
                        self.load_order.append(plugin_name)
                        print(f"✅ 加载插件: {plugin_name}")
    
    def _get_plugin_class(self, plugin_name: str):
        """动态导入插件类"""
        try:
            # 插件名称到模块的映射
            plugin_map = {
                'quality_checker': 'plugins.quality_checker.QualityCheckerPlugin',
                'deduplicator': 'plugins.deduplicator.DeduplicationPlugin',
                'monitor': 'plugins.monitor.MonitoringPlugin',
                'custom_output': 'plugins.custom_output.CustomOutputPlugin',
            }
            
            if plugin_name in plugin_map:
                module_path, class_name = plugin_map[plugin_name].rsplit('.', 1)
                module = __import__(module_path, fromlist=[class_name])
                return getattr(module, class_name)
        except Exception as e:
            print(f"❌ 加载插件 {plugin_name} 失败: {e}")
        return None
    
    def execute_pipeline(self, data: Any, context: Dict[str, Any]) -> Any:
        """执行插件流水线"""
        result = data
        
        for plugin_name in self.load_order:
            plugin = self.plugins.get(plugin_name)
            if plugin:
                try:
                    start_time = time.time()
                    result = plugin.process(result, context)
                    elapsed = time.time() - start_time
                    
                    # 记录性能指标
                    context['metrics'][f'plugin_{plugin_name}_time'] = elapsed
                    print(f"🔧 插件 {plugin_name} 处理时间: {elapsed:.3f}s")
                    
                except Exception as e:
                    if not plugin.on_error(e):
                        print(f"⚠️ 插件 {plugin_name} 错误，停止流水线")
                        break
        
        return result
    
    def cleanup_all(self) -> None:
        """清理所有插件"""
        for plugin in self.plugins.values():
            plugin.cleanup()
```

# plugins/quality_checker.py
import time
import requests
from typing import Dict, Any, Optional
from .base import Plugin

class QualityCheckerPlugin(Plugin):
    """高级质量检查插件"""
    
    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.min_response_time = config.get('min_response_time', 2000)
        self.check_availability = config.get('check_availability', True)
        self.check_resolution = config.get('check_resolution', True)
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (IPTV Quality Checker)'
        })
    
    def initialize(self) -> bool:
        print(f"🎯 初始化质量检查插件，最小响应时间: {self.min_response_time}ms")
        return True
    
    def process(self, data: Any, context: Dict[str, Any]) -> Any:
        """检查频道质量"""
        if isinstance(data, dict) and 'url' in data:
            channel_data = data
        elif isinstance(data, str) and ',' in data:
            # 处理频道行: "名称,URL"
            name, url = data.split(',', 1)
            channel_data = {'name': name, 'url': url.strip()}
        else:
            return data
        
        # 执行质量检查
        quality_score = self._check_quality(channel_data['url'])
        channel_data['quality_score'] = quality_score
        channel_data['checked_at'] = time.time()
        
        # 根据质量分数决定是否过滤
        min_score = self.config.get('min_quality_score', 60)
        if quality_score < min_score:
            print(f"⚠️  低质量源过滤: {channel_data.get('name', 'Unknown')} (分数: {quality_score})")
            return None  # 过滤掉
        
        return channel_data
    
    def _check_quality(self, url: str) -> float:
        """计算质量分数（0-100）"""
        score = 100.0
        
        try:
            # 1. 响应时间检查（权重：40%）
            start_time = time.time()
            response = self.session.head(url, timeout=5, allow_redirects=True)
            response_time = (time.time() - start_time) * 1000  # 毫秒
            
            if response_time > self.min_response_time:
                time_score = max(0, 40 - (response_time - self.min_response_time) / 50)
                score -= (40 - time_score)
            
            # 2. 状态码检查（权重：30%）
            if response.status_code != 200:
                score -= 30
            
            # 3. 内容类型检查（权重：20%）
            content_type = response.headers.get('Content-Type', '').lower()
            if 'video' not in content_type and 'audio' not in content_type:
                score -= 20
            
            # 4. 文件大小检查（权重：10%）
            content_length = response.headers.get('Content-Length')
            if content_length:
                size_mb = int(content_length) / 1024 / 1024
                if size_mb < 0.1:  # 小于100KB可能是错误页面
                    score -= 10
        
        except Exception as e:
            print(f"❌ 质量检查失败 {url}: {e}")
            score = 0.0
        
        return max(0.0, min(100.0, score))
    
    def cleanup(self) -> None:
        self.session.close()
```

# plugins/monitor.py
import time
import psutil
from datetime import datetime
from typing import Dict, Any, List
from .base import Plugin

class MonitoringPlugin(Plugin):
    """实时监控和性能分析插件"""
    
    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.metrics: Dict[str, List] = {
            'processing_times': [],
            'memory_usage': [],
            'channels_processed': [],
            'errors': []
        }
        self.start_time = time.time()
        self.process_count = 0
        self.error_count = 0
    
    def initialize(self) -> bool:
        print("📊 初始化监控插件")
        return True
    
    def process(self, data: Any, context: Dict[str, Any]) -> Any:
        """记录处理指标"""
        self.process_count += 1
        
        # 定期收集系统指标
        if self.process_count % 100 == 0:
            self._collect_system_metrics()
        
        # 如果data是None（被过滤），记录过滤
        if data is None:
            self.metrics['channels_processed'].append({
                'timestamp': datetime.now().isoformat(),
                'status': 'filtered',
                'reason': 'quality_check'
            })
        else:
            self.metrics['channels_processed'].append({
                'timestamp': datetime.now().isoformat(),
                'status': 'processed'
            })
        
        return data
    
    def _collect_system_metrics(self) -> None:
        """收集系统性能指标"""
        current_time = time.time()
        
        # CPU使用率
        cpu_percent = psutil.cpu_percent(interval=0.1)
        
        # 内存使用
        memory = psutil.virtual_memory()
        
        # 网络IO
        net_io = psutil.net_io_counters()
        
        self.metrics['processing_times'].append({
            'timestamp': datetime.fromtimestamp(current_time).isoformat(),
            'elapsed': current_time - self.start_time,
            'cpu_percent': cpu_percent,
            'memory_percent': memory.percent,
            'memory_used_mb': memory.used / 1024 / 1024,
            'bytes_sent': net_io.bytes_sent,
            'bytes_recv': net_io.bytes_recv
        })
    
    def generate_report(self) -> Dict[str, Any]:
        """生成监控报告"""
        total_time = time.time() - self.start_time
        
        return {
            'summary': {
                'total_processing_time': total_time,
                'channels_processed': self.process_count,
                'errors_count': self.error_count,
                'processing_rate': self.process_count / total_time if total_time > 0 else 0,
                'average_memory_usage': self._calculate_average('memory_percent'),
                'peak_memory_usage': self._find_peak('memory_percent'),
                'average_cpu_usage': self._calculate_average('cpu_percent')
            },
            'timeline': self.metrics['processing_times'][-100:],  # 最近100个点
            'recent_errors': self.metrics['errors'][-10:],  # 最近10个错误
            'timestamp': datetime.now().isoformat()
        }
    
    def _calculate_average(self, metric_key: str) -> float:
        """计算平均值"""
        values = [point[metric_key] for point in self.metrics['processing_times'] if metric_key in point]
        return sum(values) / len(values) if values else 0.0
    
    def _find_peak(self, metric_key: str) -> float:
        """查找峰值"""
        values = [point[metric_key] for point in self.metrics['processing_times'] if metric_key in point]
        return max(values) if values else 0.0
    
    def cleanup(self) -> None:
        """生成最终报告"""
        report = self.generate_report()
        print("📊 监控报告:")
        print(f"   总处理时间: {report['summary']['total_processing_time']:.2f}s")
        print(f"   处理频道数: {report['summary']['channels_processed']}")
        print(f"   处理速率: {report['summary']['processing_rate']:.1f} 频道/秒")
        print(f"   平均内存使用: {report['summary']['average_memory_usage']:.1f}%")
        
        # 保存报告到文件
        import json
        with open('logs/monitoring_report.json', 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False)

# core/engine.py
from typing import Dict, List, Any, Optional
import yaml
import time
from .config import ConfigManager
from .state import GlobalState
from plugins.base import PluginManager

class ProcessingEngine:
    """主处理引擎 - 协调所有组件"""
    
    def __init__(self, config_path: str = 'config.yaml'):
        # 1. 加载配置
        self.config_manager = ConfigManager(config_path)
        self.config = self.config_manager.config
        
        # 2. 初始化状态
        self.state = GlobalState(self.config)
        
        # 3. 初始化插件系统
        self.plugin_manager = PluginManager(self.config)
        self.plugin_manager.load_plugins()
        
        # 4. 初始化上下文
        self.context = {
            'start_time': time.time(),
            'config': self.config,
            'state': self.state,
            'metrics': {},
            'errors': []
        }
        
        print("🚀 处理引擎初始化完成")
    
    def run(self) -> Dict[str, Any]:
        """执行完整处理流程"""
        print("=" * 60)
        print("🎬 开始执行处理流程")
        print("=" * 60)
        
        try:
            # 阶段1: 数据采集
            self._collect_data()
            
            # 阶段2: 数据处理
            self._process_data()
            
            # 阶段3: 数据输出
            self._generate_outputs()
            
            # 阶段4: 生成报告
            report = self._generate_report()
            
            print("=" * 60)
            print("🎉 处理流程完成")
            print("=" * 60)
            
            return report
            
        except Exception as e:
            print(f"❌ 处理流程失败: {e}")
            raise
    
    def _collect_data(self) -> None:
        """数据采集阶段"""
        print("📥 阶段1: 数据采集")
        
        # 从URL采集
        urls = self.config_manager.get_urls()
        for i, url in enumerate(urls, 1):
            print(f"  📡 采集 ({i}/{len(urls)}): {url}")
            data = self._fetch_url_data(url)
            if data:
                self.state.add_raw_data(url, data)
        
        # 从文件采集
        manual_files = self.config_manager.get_manual_files()
        for file_path in manual_files:
            print(f"  📂 采集文件: {file_path}")
            data = self._read_file_data(file_path)
            if data:
                self.state.add_raw_data(file_path, data)
    
    def _process_data(self) -> None:
        """数据处理阶段"""
        print("🔧 阶段2: 数据处理")
        
        raw_data_count = len(self.state.raw_data)
        print(f"  待处理数据源: {raw_data_count}")
        
        for source, data in self.state.raw_data.items():
            print(f"  🔄 处理: {source}")
            
            # 通过插件流水线处理
            processed_data = self.plugin_manager.execute_pipeline(data, self.context)
            
            if processed_data:
                self.state.add_processed_data(source, processed_data)
    
    def _generate_outputs(self) -> None:
        """数据输出阶段"""
        print("📤 阶段3: 数据输出")
        
        output_config = self.config.get('output', {})
        
        # 生成各种格式的输出
        if output_config.get('txt', {}).get('enabled', True):
            self._generate_txt_output()
        
        if output_config.get('m3u', {}).get('enabled', True):
            self._generate_m3u_output()
        
        if output_config.get('json', {}).get('enabled', True):
            self._generate_json_output()
        
        if output_config.get('html', {}).get('enabled', True):
            self._generate_html_output()
    
    def _generate_report(self) -> Dict[str, Any]:
        """生成处理报告"""
        print("📊 阶段4: 生成报告")
        
        # 收集所有指标
        report = {
            'execution': {
                'start_time': self.context['start_time'],
                'end_time': time.time(),
                'duration': time.time() - self.context['start_time'],
                'sources_processed': len(self.state.raw_data),
                'channels_processed': len(self.state.processed_data),
            },
            'metrics': self.context['metrics'],
            'plugin_metrics': {},
            'system_info': self._get_system_info(),
            'config_summary': self._get_config_summary()
        }
        
        # 保存报告
        import json
        report_file = f"logs/execution_report_{int(time.time())}.json"
        with open(report_file, 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False)
        
        print(f"  ✅ 报告已保存: {report_file}")
        return report

# core/output_manager.py
import os
import json
import gzip
from datetime import datetime
from typing import Dict, List, Any, Optional

class OutputManager:
    """智能输出管理器"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.output_dir = config.get('output_dir', 'output')
        os.makedirs(self.output_dir, exist_ok=True)
        
        # 输出模板
        self.templates = self._load_templates()
    
    def _load_templates(self) -> Dict[str, Any]:
        """加载输出模板"""
        templates_dir = 'templates'
        templates = {}
        
        if os.path.exists(templates_dir):
            for file in os.listdir(templates_dir):
                if file.endswith('.json'):
                    template_name = file[:-5]
                    with open(os.path.join(templates_dir, file), 'r', encoding='utf-8') as f:
                        templates[template_name] = json.load(f)
        
        return templates
    
    def generate_output(self, 
                       data: List[Dict[str, Any]], 
                       output_type: str,
                       template: str = 'default') -> str:
        """生成指定类型的输出"""
        
        if output_type == 'txt':
            return self._generate_txt(data, template)
        elif output_type == 'm3u':
            return self._generate_m3u(data, template)
        elif output_type == 'json':
            return self._generate_json(data, template)
        elif output_type == 'html':
            return self._generate_html(data, template)
        else:
            raise ValueError(f"不支持的输出类型: {output_type}")
    
    def _generate_txt(self, data: List[Dict[str, Any]], template: str) -> str:
        """生成TXT格式"""
        output_lines = []
        
        # 按分类分组
        categories = {}
        for item in data:
            category = item.get('category', '其他')
            if category not in categories:
                categories[category] = []
            categories[category].append(item)
        
        # 生成输出
        for category_name, items in categories.items():
            # 分组标题
            emoji = self.config['channels']['categories'].get(category_name, {}).get('emoji', '📺')
            output_lines.append(f"{emoji}{category_name},#genre#")
            
            # 频道列表
            for item in items:
                name = item.get('name', '未知')
                url = item.get('url', '')
                quality = item.get('quality_score', 0)
                
                # 添加质量标记
                if quality >= 90:
                    name = f"{name} 🔥"
                elif quality >= 70:
                    name = f"{name} ⚡"
                
                output_lines.append(f"{name},{url}")
            
            output_lines.append("")  # 空行分隔
        
        return "\n".join(output_lines)
    
    def _generate_m3u(self, data: List[Dict[str, Any]], template: str) -> str:
        """生成M3U格式"""
        output_lines = ['#EXTM3U']
        
        # EPG配置
        epg_url = self.config['output']['m3u'].get('epg_url', '')
        if epg_url:
            output_lines.append(f'#EXTM3U x-tvg-url="{epg_url}"')
        
        # 频道列表
        for item in data:
            name = item.get('name', '未知')
            url = item.get('url', '')
            logo = item.get('logo', '')
            category = item.get('category', '其他')
            group = item.get('group', category)
            
            # EXTINF行
            duration = '-1'
            if logo:
                output_lines.append(f'#EXTINF:{duration} tvg-name="{name}" tvg-logo="{logo}" group-title="{group}",{name}')
            else:
                output_lines.append(f'#EXTINF:{duration} group-title="{group}",{name}')
            
            # URL行
            output_lines.append(url)
        
        return "\n".join(output_lines)
    
    def save_output(self, content: str, filename: str, compress: bool = False) -> str:
        """保存输出到文件"""
        filepath = os.path.join(self.output_dir, filename)
        
        if compress:
            filepath += '.gz'
            with gzip.open(filepath, 'wt', encoding='utf-8') as f:
                f.write(content)
        else:
            with open(filepath, 'w', encoding='utf-8') as f:
                f.write(content)
        
        # 记录文件信息
        file_size = os.path.getsize(filepath) / 1024  # KB
        print(f"  💾 保存文件: {filename} ({file_size:.1f}KB)")
        
        return filepath
```
