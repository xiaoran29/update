🎨 第四阶段：名称处理细节

频道名称清理流程：

```
输入：CCTV-1 HD「IPV6」_电信
1. 清理特定字符 → CCTV-1 HD
2. 移除末尾HD → CCTV-1
3. 繁体转简体 → CCTV-1（无变化）
4. 特殊处理CCTV → CCTV-1（标准化）
```

CCTV名称标准化：

```python
def process_part(part_str):
    if "CCTV" in part_str and "://" not in part_str:
        part_str = part_str.replace("IPV6", "").replace("PLUS", "+").replace("1080", "")
        filtered_str = ''.join(char for char in part_str if char.isdigit() or char == 'K' or char == '+')
        if not filtered_str.strip():
            filtered_str = part_str.replace("CCTV", "")
        return "CCTV" + filtered_str
```

```python
def get_http_response(url, timeout=8, retries=2, backoff_factor=1.0):
    """带重试机制的HTTP请求"""
    for attempt in range(retries):
        try:
            req = urllib.request.Request(url, headers=headers)
            with urllib.request.urlopen(req, timeout=timeout) as response:
                return response.read().decode('utf-8')
        except Exception as e:
            if attempt < retries - 1:
                time.sleep(backoff_factor * (2 ** attempt))
    return None
```

```python
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
```

```python
# 添加性能监控装饰器
import time
from functools import wraps

def monitor_performance(metric_name):
    """性能监控装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            start_time = time.time()
            result = func(*args, **kwargs)
            end_time = time.time()
            
            duration = end_time - start_time
            
            # 记录到状态
            if hasattr(args[0], 'state'):
                args[0].state.stats[f'{metric_name}_time'] = duration
            
            print(f"⏱️  {metric_name}: {duration:.2f}秒")
            return result
        return wrapper
    return decorator

# 使用示例

    """处理所有URL"""
```

```python
def health_check(state):
    """系统健康检查"""
    
    checks = []
    
    # 检查必要文件
    required_files = [
        Config.URLS_FILE,
        Config.CORRECTIONS_FILE,
        Config.LOGO_FILE,
    ]
    
    for file_path in required_files:
        if os.path.exists(file_path):
            checks.append(f"✅ {os.path.basename(file_path)} 存在")
        else:
            checks.append(f"❌ {os.path.basename(file_path)} 缺失")
    
    # 检查输出目录
    if os.path.exists(Config.OUTPUT_DIR):
        checks.append(f"✅ 输出目录存在")
    else:
        checks.append(f"⚠️  输出目录不存在，将自动创建")
    
    # 检查内存使用
    import psutil
    memory_usage = psutil.Process().memory_info().rss / 1024 / 1024
    checks.append(f"📊 内存使用: {memory_usage:.1f}MB")
    
    # 打印检查结果
    print("🔍 系统健康检查:")
    for check in checks:
        print(f"  {check}")
    
    return all('✅' in check or '⚠️' in check for check in checks)
```

2. 🔧 错误处理最佳实践

```python
class ProcessingError(Exception):
    """处理错误基类"""
    pass

class URLFetchError(ProcessingError):
    """URL获取错误"""
    pass

class ChannelParseError(ProcessingError):
    """频道解析错误"""
    pass

def process_with_error_handling(func):
    """带错误处理的装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except URLFetchError as e:
            print(f"❌ URL获取失败: {e}")
            # 重试或跳过
            return None
        except ChannelParseError as e:
            print(f"❌ 频道解析失败: {e}")
            # 记录到错误日志
            log_error(e)
            return None
        except Exception as e:
            print(f"❌ 未知错误: {e}")
            raise
    return wrapper
```
